Option Explicit

' MSForms の定数(参照設定に頼らないため、値をここで宣言する)
Private Const fmShiftMask As Long = 1

'--- ユーザ名: Enter / Tab → パスワードへ ---
Private Sub txtUserName_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Select Case KeyCode
        Case vbKeyReturn
            KeyCode = 0
            Me.OLEObjects("txtPassword").Activate
        Case vbKeyTab
            If (Shift And fmShiftMask) = 0 Then
                KeyCode = 0
                Me.OLEObjects("txtPassword").Activate
            End If
    End Select
End Sub

'--- パスワード: Enter → ログイン、Tab → ログインボタン、Shift+Tab → ユーザ名 ---
Private Sub txtPassword_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Select Case KeyCode
        Case vbKeyReturn
            KeyCode = 0
            modMain.DoLogin
        Case vbKeyTab
            KeyCode = 0
            If (Shift And fmShiftMask) <> 0 Then
                Me.OLEObjects("txtUserName").Activate
            Else
                Me.OLEObjects("cmdLogin").Activate
            End If
    End Select
End Sub

'--- ログインボタン: Enter → ログイン、Shift+Tab → パスワード ---
Private Sub cmdLogin_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    Select Case KeyCode
        Case vbKeyReturn
            KeyCode = 0
            modMain.DoLogin
        Case vbKeyTab
            If (Shift And fmShiftMask) <> 0 Then
                KeyCode = 0
                Me.OLEObjects("txtPassword").Activate
            End If
    End Select
End Sub

Private Sub cmdLogin_Click()
    modMain.DoLogin
End Sub
