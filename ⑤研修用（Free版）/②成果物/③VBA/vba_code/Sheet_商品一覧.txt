'==============================================================
' 「商品一覧」シートのコード
' VBE で「商品一覧」シートのコード画面を全選択して削除し、
' この内容(この行以降すべて)を貼り付けてください。
'==============================================================
Option Explicit

' 「注文状況」(F 列)のハイパーリンクをクリックしたら、その行の商品コードの注文を表示する
Private Sub Worksheet_FollowHyperlink(ByVal Target As Hyperlink)
    Dim r As Long
    Dim productCode As String

    On Error GoTo EH

    r = Target.Range.Row
    If r < 4 Or Target.Range.Column <> 6 Then Exit Sub

    productCode = Trim$(CStr(Me.Cells(r, 1).Value))
    If Len(productCode) = 0 Then Exit Sub

    modMain.ShowOrderDetail productCode
    Exit Sub

EH:
    MsgBox "注文情報詳細の表示でエラーが発生しました。" & vbCrLf & vbCrLf & _
           "プロシージャ: 商品一覧.Worksheet_FollowHyperlink" & vbCrLf & _
           "エラー番号: " & CStr(Err.Number) & vbCrLf & _
           "内容: " & Err.Description, vbExclamation, "エラー"
End Sub
